cmake_minimum_required(VERSION 3.20)
# The one version number of the Windows, Android and PS5 apps.
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" ARCHIVEFETCH_VERSION LIMIT_COUNT 1)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
string(REGEX MATCH "^[0-9]+\\.[0-9]+\\.[0-9]+" ARCHIVEFETCH_VERSION_NUMBER "${ARCHIVEFETCH_VERSION}")
project(archive_catalog VERSION ${ARCHIVEFETCH_VERSION_NUMBER} LANGUAGES CXX)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS VERSION)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
# Some Windows CMake builds (e.g. Strawberry's, GnuTLS) ship without a CA bundle and reject every HTTPS download;
# give them the certificates Windows itself trusts.
if(WIN32 AND NOT CMAKE_TLS_CAINFO AND NOT DEFINED ENV{SSL_CERT_FILE})
    set(windows_roots "${CMAKE_BINARY_DIR}/windows-roots.pem")
    execute_process(COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/cmake/export-windows-roots.ps1"
        -Output "${windows_roots}" RESULT_VARIABLE roots_result ERROR_VARIABLE roots_error)
    if(roots_result EQUAL 0)
        set(CMAKE_TLS_CAINFO "${windows_roots}")
    else()
        message(WARNING "Could not export the Windows trusted certificates; HTTPS downloads may fail: ${roots_result} ${roots_error}")
    endif()
endif()

find_package(LibXml2 REQUIRED)
add_library(catalog src/catalog.cpp)
target_include_directories(catalog PUBLIC src)
target_link_libraries(catalog PUBLIC LibXml2::LibXml2)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(catalog PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

enable_testing()
add_executable(catalog_tests tests/catalog_tests.cpp)
target_link_libraries(catalog_tests PRIVATE catalog)
add_test(NAME catalog_tests COMMAND catalog_tests)

            URL https://curl.se/download/curl-8.18.0.tar.xz

    add_library(transfer src/transfer.cpp)

    add_executable(archive_catalog src/main.cpp)
    target_link_libraries(archive_catalog PRIVATE application)

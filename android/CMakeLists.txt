cmake_minimum_required(VERSION 3.22)
project(archive_android LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_OBJECT_PATH_MAX 200)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
include(FetchContent)
FetchContent_Declare(boringssl
    URL https://github.com/google/boringssl/archive/refs/tags/0.20261020.0.tar.gz
    URL_HASH SHA256=fb43aac7fbd323c9dfa1f45fe92a1c01142433c05b23fe34f61d6ba8b13b642e)
FetchContent_GetProperties(boringssl)
if(NOT boringssl_POPULATED)
    FetchContent_Populate(boringssl)
endif()
execute_process(COMMAND "${CMAKE_COMMAND}" -S "${boringssl_SOURCE_DIR}" -B "${boringssl_BINARY_DIR}" -G Ninja
    "-DCMAKE_TOOLCHAIN_FILE=${CMAKE_TOOLCHAIN_FILE}" "-DCMAKE_MAKE_PROGRAM=${CMAKE_MAKE_PROGRAM}"
    "-DANDROID_ABI=${ANDROID_ABI}" "-DANDROID_PLATFORM=${ANDROID_PLATFORM}" "-DANDROID_STL=c++_shared"
    -DCMAKE_BUILD_TYPE=Release -DCMAKE_POSITION_INDEPENDENT_CODE=ON
    RESULT_VARIABLE tls_configured)
if(NOT tls_configured EQUAL 0)
    message(FATAL_ERROR "Android TLS configuration failed")
endif()
execute_process(COMMAND "${CMAKE_COMMAND}" --build "${boringssl_BINARY_DIR}" --target ssl crypto --parallel 4
    RESULT_VARIABLE tls_built)
if(NOT tls_built EQUAL 0)
    message(FATAL_ERROR "Android TLS build failed")
endif()
list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
foreach(feature ICONV ZLIB LZMA PYTHON TESTS PROGRAMS HTTP FTP)
    set(LIBXML2_WITH_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
FetchContent_Declare(xml
    URL https://download.gnome.org/sources/libxml2/2.15/libxml2-2.15.3.tar.xz
    URL_HASH SHA256=78262a6e7ac170d6528ebfe2efccdf220191a5af6a6cd61ea4a9a9a5042c7a07)
FetchContent_MakeAvailable(xml)
foreach(feature BUILD_CURL_EXE BUILD_LIBCURL_DOCS BUILD_MISC_DOCS CURL_USE_LIBPSL CURL_ZLIB CURL_BROTLI CURL_ZSTD CURL_USE_LIBSSH2 CURL_USE_LIBSSH CURL_USE_GSSAPI CURL_USE_LIBIDN2)
    set(${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(HTTP_ONLY ON CACHE BOOL "" FORCE)
set(CURL_USE_OPENSSL ON CACHE BOOL "" FORCE)
set(CURL_CA_BUNDLE none CACHE STRING "" FORCE)
set(CURL_CA_PATH none CACHE STRING "" FORCE)
FetchContent_Declare(curl URL https://curl.se/download/curl-8.18.0.tar.xz
    URL_HASH SHA256=40df79166e74aa20149365e11ee4c798a46ad57c34e4f68fd13100e2c9a91946)
FetchContent_MakeAvailable(curl)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz
    URL_HASH SHA256=d6c65aca6b1ed68e7a182f4757257b107ae403032760ed6ef121c9d55e81757d)
FetchContent_MakeAvailable(json)
set(SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../src")
add_library(archive_android SHARED
    "${SOURCE}/web_main.cpp" "${SOURCE}/catalog.cpp" "${SOURCE}/transfer.cpp"
    "${SOURCE}/application.cpp" "${SOURCE}/gui_model.cpp" "${SOURCE}/gui_platform.cpp")
target_include_directories(archive_android PRIVATE "${SOURCE}")
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION" ARCHIVEFETCH_VERSION LIMIT_COUNT 1)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION")
target_compile_definitions(archive_android PRIVATE ARCHIVEFETCH_VERSION="${ARCHIVEFETCH_VERSION}")
target_link_libraries(archive_android PRIVATE LibXml2::LibXml2 CURL::libcurl OpenSSL::Crypto nlohmann_json::nlohmann_json log)
target_link_options(archive_android PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
if(NOT DEFINED ANDROID_LICENSE_DIR)
    set(ANDROID_LICENSE_DIR "${CMAKE_CURRENT_BINARY_DIR}/licenses")
endif()
file(MAKE_DIRECTORY "${ANDROID_LICENSE_DIR}")
foreach(dependency boringssl xml curl json)
    if(dependency STREQUAL "json")
        set(license LICENSE.MIT)
    elseif(dependency STREQUAL "xml")
        set(license Copyright)
    elseif(dependency STREQUAL "curl")
        set(license COPYING)
    else()
        set(license LICENSE)
    endif()
    configure_file("${${dependency}_SOURCE_DIR}/${license}" "${ANDROID_LICENSE_DIR}/${dependency}.txt" COPYONLY)
endforeach()
